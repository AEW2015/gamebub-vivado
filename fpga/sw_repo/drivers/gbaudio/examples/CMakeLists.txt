# BSP Import Examples build: Vitis supplies instance/example metadata and linker.
# Building an ELF never runs it; review each example's safety settings first.
# License details: LICENSE.md at repository root.
cmake_minimum_required(VERSION 3.15)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/GbaudioExample.cmake")
  include(${CMAKE_CURRENT_SOURCE_DIR}/GbaudioExample.cmake)
  project(gbaudio C ASM)
  find_package(common)
  collect(PROJECT_LIB_DEPS xilstandalone)
  collect(PROJECT_LIB_DEPS xil)
  collect(PROJECT_LIB_DEPS gcc)
  collect(PROJECT_LIB_DEPS c)
  collector_list(_deps PROJECT_LIB_DEPS)
  set(DRIVER_INSTANCES "${NUM_DRIVER_INSTANCES}" CACHE STRING "Driver Instances")
  set_property(CACHE DRIVER_INSTANCES PROPERTY STRINGS "${NUM_DRIVER_INSTANCES}")
  set(index 0)
  LIST_INDEX(${index} ${DRIVER_INSTANCES} "${NUM_DRIVER_INSTANCES}")
  list(GET TOTAL_EXAMPLE_LIST ${index} ex_list)
  set(EXAMPLE_LIST "${${ex_list}}" CACHE STRING "Driver Example List")
  set_property(CACHE EXAMPLE_LIST PROPERTY STRINGS "${${ex_list}}")
  linker_gen("${CMAKE_CURRENT_SOURCE_DIR}/linker_files/")
  foreach(example IN LISTS EXAMPLE_LIST)
    string(REGEX REPLACE "\\.[^.]*$" "" stem "${example}")
    add_executable(${stem}.elf ${example})
    target_link_libraries(${stem}.elf -Wl,--gc-sections -T\"${CMAKE_SOURCE_DIR}/lscript.ld\" -L\"${CMAKE_SOURCE_DIR}/\" -Wl,--start-group ${_deps} -Wl,--end-group)
  endforeach()
else()
  project(gbaudio_examples C)
  add_library(gbaudio_examples STATIC audio_tone.c audio_safety.c)
  target_include_directories(gbaudio_examples PUBLIC ../src)
  target_compile_definitions(gbaudio_examples PRIVATE TESTAPP_GEN)
  target_link_libraries(gbaudio_examples PUBLIC gbaudio)
endif()
